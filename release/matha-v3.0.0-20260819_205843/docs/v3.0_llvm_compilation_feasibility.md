# Matha v3.0 — LLVM IR 编译目标可行性分析

> 分析将 Matha 编译目标从 Python 解释执行改为 LLVM IR 的可行性、改造步骤与收益。

## 1. 现状分析

### 1.1 当前执行模型

```
┌─────────────────────────────────────────────────────────────┐
│                    当前执行链路                              │
│                                                             │
│  自然语言 → IntentParser → MIR → Python 代码 → exec()       │
│             (正则/LLM)         (字符串模板)    (解释执行)      │
│                                                             │
│  性能瓶颈：                                                  │
│  1. Python GIL 限制并发                                     │
│  2. exec() 动态执行，无编译优化                              │
│  3. 无类型信息，无法 SIMD 向量化                             │
│  4. 每次执行重新解析，无 JIT 缓存                            │
└─────────────────────────────────────────────────────────────┘
```

### 3.2 核心改造步骤

#### Step 1: 统一 IR 接口

```python
# src/compiler/ir.py — 新增统一 IR 接口
from enum import Enum
from dataclasses import dataclass, field
from typing import Any, Optional

class IRBackend(Enum):
    PYTHON = "python"    # 解释执行（默认）
    LLVM = "llvm"        # LLVM 编译（高性能）
    C = "c"             # C 代码生成（跨平台）

@dataclass
class CompileTarget:
    """编译目标配置。"""
    backend: IRBackend = IRBackend.PYTHON
    optimize: bool = True
    cache_dir: str = ".matha_cache"
    llvm_flags: list[str] = field(default_factory=lambda: ["-O2"])

class IRCompiler:
    """统一 IR 编译器接口。"""

    def compile(self, mir_code: str, target: CompileTarget) -> CompileResult:
        """编译 MIR 代码到目标后端。"""
        if target.backend == IRBackend.PYTHON:
            return self._compile_python(mir_code, target)
        elif target.backend == IRBackend.LLVM:
            return self._compile_llvm(mir_code, target)
        elif target.backend == IRBackend.C:
            return self._compile_c(mir_code, target)
        else:
            raise ValueError(f"不支持的后端: {target.backend}")
```

#### Step 2: 集成 LLVM 后端

```python
# src/compiler/ir.py — 扩展实现

from src.compiler.llvm_backend import LLVMCompiler

class IRCompiler:
    def __init__(self):
        self._llvm_compiler = LLVMCompiler()
        self._cache = {}

    def _compile_llvm(self, mir_code: str, target: CompileTarget) -> CompileResult:
        """编译到 LLVM IR → 机器码。"""
        # 1. 生成 LLVM IR
        llvm_ir = self._mir_to_llvm(mir_code)

        # 2. 缓存检查
        cache_key = hashlib.sha256(llvm_ir.encode()).hexdigest()[:16]
        if target.optimize and cache_key in self._cache:
            return self._cache[cache_key]

        # 3. 调用 LLVM 编译器
        compiler = self._llvm_compiler
        result = compiler.compile(llvm_ir, func_name="main")

        compile_result = CompileResult(
            backend=IRBackend.LLVM,
            code=llvm_ir,
            executable=result,
            cache_key=cache_key,
        )

        if target.optimize:
            self._cache[cache_key] = compile_result

        return compile_result

    def _mir_to_llvm(self, mir_code: str) -> str:
        """MIR → LLVM IR 转换。"""
        # 复用现有的 llvm_backend.py 逻辑
        from src.compiler.llvm_backend import MathaIRToLLVM
        converter = MathaIRToLLVM()
        return converter.generate(mir_code)
```

```python
# src/enhanced_intent.py — 集成编译目标选择

class EnhancedIntentParser(IntentParserBase):
    def __init__(self, llm_parser=None, compile_target: CompileTarget = None):
        super().__init__()
        self.llm_parser = llm_parser
        self.compile_target = compile_target or CompileTarget()
        self._compiler = IRCompiler()

    def parse(self, text: str, target_lang: str = "mir") -> Result:
        # ... 原有解析逻辑 ...

        # 新增：根据编译目标选择执行方式
        if self.compile_target.backend == IRBackend.LLVM:
            compile_result = self._compiler.compile(intent.suggested_code, self.compile_target)
            intent.executable = compile_result.executable
            intent.exec_mode = "compiled"
        else:
            intent.exec_mode = "interpreted"

        return Ok(intent)

    def execute(self, intent: Intent) -> Result:
        """执行意图，支持编译/解释两种模式。"""
        if intent.exec_mode == "compiled" and intent.executable:
            # 执行编译后的函数
            try:
                result = intent.executable()
                return Ok(result)
            except Exception as e:
                return Err(ExecError(f"编译代码执行失败: {e}"))
        else:
            # 解释执行（原有逻辑）
            return self._execute_interpreted(intent)
```
